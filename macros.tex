% Register a notation for the Index of Notations (no inline display).

% Global counter: incremented on every \regnotation call, never reset.
% Its value (padded to three digits) forms the within-chapter order in the sort key.
\newcounter{regnotationcnt}

% Internal: expands to the automatic sort key — chapter letter (A, B, C, …)
% followed by the current counter value zero-padded to three digits.
% Example: second notation called while in chapter 3 → "C002".
% The leading letter is the group key consumed by \nomgroup.
\newcommand{\regnotautokey}{%
  \Alph{chapter}%
  \ifnum\value{regnotationcnt}<10 00\else
  \ifnum\value{regnotationcnt}<100 0\fi\fi
  \arabic{regnotationcnt}%
}

% \regnotation{symbol}
% Call on the FIRST occurrence of each notation in the text.
% Displays the symbol inline and registers it in the List of Notations
% with its chapter group and a hyperlinked page number.
% An explicit sort key can be passed as an optional first argument
% (e.g.\ \regnotation[Z99]{...}) as an escape hatch.
%
% \regnotationsym{symbol} — display-only variant, no side effects.
% Used internally by \denval (and similar) to avoid double-registration
% when the term argument is re-expanded for subscripts/types.
\newcommand{\regnotationsym}[2][]{\ensuremath{#2}}
\newcommand{\regnotation}[2][]{%
  \stepcounter{regnotationcnt}%
  \def\regnotsortkey{#1}%
  \ifx\regnotsortkey\empty
    \def\regnotsortkey{\regnotautokey}%
  \fi
  \edef\regnotsortkey{\regnotsortkey}%  % expand to a plain string (e.g. "B003")
  \hypertarget{notat:\regnotsortkey}{}% % PDF anchor at the call site
  \ensuremath{#2}%
  \nomenclature[\regnotsortkey]{$#2$}{}%
}

% ── Hyperlink plumbing for the List of Notations ─────────────────────────────
%
% \@@@nomenclature is nomencl's internal writer (called by \nomenclature after
% \makenomenclature).  We patch it to embed the sort key as \nomeqref's argument
% instead of \theequation.  This gives us a side-channel to carry the sort key
% into the .nls typesetting phase, where \nomeqref stores it globally and
% \pagedeclaration uses it to build a \hyperlink back to the call site.
\makeatletter
\def\@@@nomenclature[#1]#2#3{%
  \def\@tempa{#2}\def\@tempb{#3}%
  \protected@write\@nomenclaturefile{}%
    {\string\nomenclatureentry{#1\nom@verb\@tempa @[{\nom@verb\@tempa}]%
        \begingroup\nom@verb\@tempb\protect\nomeqref{#1}%
        |nompageref}{\thepage}}%
  \endgroup
  \@esphack}
\def\nom@currentlinkkey{}
\renewcommand{\nomeqref}[1]{%
  \gdef\nom@currentlinkkey{#1}%
  \if@printeqref\eqdeclaration{#1}\fi
  \ignorespaces}
\makeatother

% preventing double-encoding of Unicode declaration names (e.g. π₁ → %CF%80%E2%82%81).
\begin{luacode*}
function pct_encode(s)
  return (s:gsub("([^\32-\126])", function(c)
    return string.format("%%%02X", string.byte(c))
  end))
end
\end{luacode*}
