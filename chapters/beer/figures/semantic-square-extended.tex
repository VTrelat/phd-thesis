\begin{tikzpicture}[
    transform shape,
    every node/.style={
      align=center,
      text=dark-blue,
      font=\normalsize,
    },
    typed/.style={
      draw=dark-blue,
      rounded corners=2mm,
      fill=beige!35,
      minimum height=8mm,
      inner xsep=6pt,
      inner ysep=5pt,
    },
    instantiated/.style={
      draw=blue!70!dark-blue,
      rounded corners=2mm,
      fill=blue!7,
      minimum height=8mm,
      inner xsep=6pt,
      inner ysep=5pt,
    },
    denotation/.style={
      draw=dark-blue,
      rounded corners=2mm,
      fill=orange!14,
      minimum height=8mm,
      inner xsep=6pt,
      inner ysep=5pt,
    },
    construction/.style={
      funarr,
      draw=blue,
    },
    consequence/.style={
      funarr,
      draw=blue,
      dash pattern=on 4pt off 2pt,
    },
    relation/.style={
      {Stealth[length=6pt, width=5pt]}-{Stealth[length=6pt, width=5pt]},
      draw=pink,
      dash pattern=on 4pt off 2pt,
      line width=1.1pt,
      shorten >=3pt,
      shorten <=3pt,
    },
    arrow label/.style={
      fill=white,
      font=\small,
      inner sep=1.5pt,
    },
  ]
  % Open, well-typed terms.
  \node[
    typed,
    label={
      [font=\small, text=dark-blue, label distance=2mm]
      above:{\(\btype{\Gamma}{t}{\alpha}\)}
    },
  ] (b-term)
    {\(t\)};
  \node[
    typed,
    right=30mm of b-term,
    label={
      [font=\small, text=dark-blue, label distance=2mm]
      above:{\(
        \begin{gathered}
          \stype{\Lambda'}{t'}{\sigma}
          \\[-1mm]
        \sigma\in
        [\lgf{(\toSMT{\alpha})},\mgf{(\toSMT{\alpha})}]
        \end{gathered}
      \)}
    },
  ] (s-term)
    {\((t',\sigma)\)};
  \node[
    typed,
    right=36mm of s-term,
    label={
      [font=\small, text=dark-blue, label distance=2mm]
      above:{\(
        \begin{aligned}
          \stype{\Lambda'[\lvar{z}\mapsto\toSMT{\alpha}]&}{\lvar{z}}{\toSMT{\alpha}}
          \\[-1mm]
          \stype{\Lambda'[\lvar{z}\mapsto\toSMT{\alpha}]&}{\spec{\lvar{z}}}{\boolS}
        \end{aligned}
      \)}
    },
  ] (loosened)
    {\(\langle\lvar{z},\spec{\lvar{z}}\rangle\)};

  \draw[construction]
    (b-term) --
    node[arrow label, above=2pt]
      {\(\encto\)}
    (s-term);
  \draw[construction]
    (s-term) --
    node[arrow label, above=2pt]
      {\(\mathsf{loosen}_{\mathbb c}\)}
    node[arrow label, below=2pt]
      {\(\mathbb c:\castpath{\sigma}{\toSMT{\alpha}}\)}
    (loosened);

  % First apply the valuations: this is the first half of
  % \interp{\aterm{\cdot}{\cdot}}.
  \node[instantiated, below=18mm of b-term] (b-instance)
    {\(\aterm{t}{\Delta}\)};
  \node[instantiated, below=18mm of s-term] (s-instance)
    {\(\aterm{t'}{\Theta'}\)};
  \coordinate (z-column) at ($(loosened.south)+(-16mm,0)$);
  \coordinate (spec-column) at ($(loosened.south)+(16mm,0)$);
  \node[instantiated] (z-instance) at (z-column |- s-instance)
    {\(\aterm{\lvar{z}}{\Theta^!}\)};
  \node[instantiated] (spec-instance) at (spec-column |- s-instance)
    {\(\aterm{\spec{\lvar{z}}}{\Theta^!}\)};

  \draw[construction]
    (b-term) --
    node[arrow label, left=2pt]
      {\(\aterm{\cdot}{\Delta}\)}
    (b-instance);
  \draw[consequence]
    (s-term) --
    node[arrow label, left=2pt]
      {\(\aterm{\cdot}{\Theta'}\)}
    (s-instance);
  \draw[consequence]
    (loosened) --
    node[arrow label, pos=.58, left=1pt]
      {\(\aterm{\cdot}{\Theta^!}\)}
    (z-instance);
  \draw[consequence]
    (loosened) --
    node[arrow label, pos=.58, right=1pt]
      {\(\aterm{\cdot}{\Theta^!}\)}
    (spec-instance);

  % Then interpret the resulting closed terms: this is the second half.
  \node[denotation, below=18mm of b-instance] (T)
    {\(\denval{T}[\alpha]\)};
  \node[denotation, below=18mm of s-instance] (T-prime)
    {\(\denval{T'}[\sigma]\)};
  \node[denotation] (T-cast) at (z-instance |- T-prime)
    {\(\denval{T^!}[\toSMT{\alpha}]\)};
  \node[denotation] (spec-true) at (spec-instance |- T-prime)
    {\(\denval{\topZ}[\boolS]\)};

  \draw[construction]
    (b-instance) --
    node[arrow label, left=2pt]
      {\(\interpB{\cdot}\)}
    (T);
  \draw[consequence]
    (s-instance) --
    node[arrow label, left=2pt]
      {\(\interpS{\cdot}\)}
    (T-prime);
  \draw[consequence]
    (z-instance) --
    node[arrow label, left=2pt]
      {\(\interpS{\cdot}\)}
    (T-cast);
  \draw[consequence]
    (spec-instance) --
    node[arrow label, right=2pt]
      {\(\interpS{\cdot}\)}
    (spec-true);

  % The lower edge of the semantic square, factored through the
  % canonical SMT representation.
  \draw[relation]
    (T) --
    node[arrow label, text=pink, above=2pt]
      {\(\semrel\)}
    (T-prime);
  \draw[consequence]
    (T-prime) --
    node[arrow label, above=2pt]
      {\(\appZ\zop{\mathbb c}(\,\cdot\,)\)}
    (T-cast);

  \draw[consequence, draw=pink]
    (T-cast.south) to[out=-120, in=-60, looseness=.65]
    node[arrow label, pos=.5, above=4pt, text=pink]
      {\(\appZ\retr{\alpha}(T^!)=T\)}
    (T.south);
\end{tikzpicture}
