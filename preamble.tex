% ---------------------------------------------------------------------------
% A5: fitting A4-tuned material into a narrower measure
% ---------------------------------------------------------------------------
% The tables, displays, grammars and figures of this manuscript were written
% against the A4 text block, 159mm wide.  The A5 one is 116mm, so some of them
% no longer fit -- by up to 43mm, which is just the difference between the two
% measures.  Rather than rewrite the manuscript for one target, shrink
% whatever overflows, and only there: on A4 none of this is loaded and the
% output is unchanged.

% A few pictures are positioned against the page rather than the text block --
% the cover artwork, the Venn chapter heads -- and must never be scaled to the
% measure.  They clear this switch around themselves, so it has to exist for
% every target, not just A5.
\newif\ifthesisfitpics \thesisfitpicstrue

\usepackage{adjustbox}

\ifThesisAfive
  % Scales an over-wide equation or gather display down to the measure and
  % leaves the ones that already fit alone.
  \usepackage{resizegather}

  % Let TeX loosen a paragraph rather than overrun the margin when a line ends
  % up holding something it cannot break -- a long Lean identifier, a URL, a
  % wide inline formula.  This settles most of the prose overflows on its own.
  \emergencystretch=3em

  % A display environment needs vertical mode and so cannot be put in a box --
  % but its contents can.  Scaling those keeps the display's own skips and
  % centring while bringing an over-wide formula back inside the measure; one
  % that already fits comes through untouched, and \tag still reaches the outer
  % display and prints flush right.  Patching equation* covers \[ ... \] too,
  % which is what amsmath makes it.  align needs vertical mode all the way down
  % and cannot be treated this way; the few that overflow use the fitalign
  % environment defined below instead.
  \makeatletter
  \expandafter\let\expandafter\thesisorigeqstar\csname equation*\endcsname
  \expandafter\let\expandafter\thesisorigendeqstar\csname endequation*\endcsname
  \expandafter\def\csname equation*\endcsname{%
    \thesisorigeqstar\adjustbox{max width=\linewidth}\bgroup$\displaystyle}
  \expandafter\def\csname endequation*\endcsname{%
    $\egroup\thesisorigendeqstar}
  \makeatother

  % Long dotted or underscored Lean names cannot be broken inside \texttt and
  % overrun the measure; give them break points at the underscores.
  \newcommand{\ThesisBreakUnderscore}{\renewcommand{\_}{\textunderscore\allowbreak}}

  % Shrink-to-measure wrappers, with a depth counter apiece so that a nested
  % occurrence does not get a second box of its own.
  \newcommand{\thesisfitopen}{\begin{adjustbox}{max width=\linewidth}}
  \newcommand{\thesisfitclose}{\end{adjustbox}}
  \newcommand{\thesisfitnext}{}

  % Wide tables: pgfplotstable's generated ones in the evaluation chapter.
  \newcounter{thesisfittabdepth}
  \BeforeBeginEnvironment{tabular}{%
    \stepcounter{thesisfittabdepth}%
    \ifnum\value{thesisfittabdepth}=1
      \let\thesisfitnext\thesisfitopen
    \else
      \let\thesisfitnext\relax
    \fi
    \thesisfitnext}
  \AfterEndEnvironment{tabular}{%
    \ifnum\value{thesisfittabdepth}=1
      \let\thesisfitnext\thesisfitclose
    \else
      \let\thesisfitnext\relax
    \fi
    \thesisfitnext
    \addtocounter{thesisfittabdepth}{-1}}

  % simplebnf builds its grammars on tabularray rather than tabular, so the
  % hook above never sees them; the syntax chapters set them at the full A4
  % measure.
  \BeforeBeginEnvironment{bnf}{\thesisfitopen}
  \AfterEndEnvironment{bnf}{\thesisfitclose}

  % Figures drawn at absolute coordinates keep their A4 width even after being
  % re-rendered for A5.  Page-anchored pictures opt out through
  % \thesisfitpicsfalse; overlay pictures have no width to scale anyway.
  \newcounter{thesisfitpicdepth}
  \BeforeBeginEnvironment{tikzpicture}{%
    \stepcounter{thesisfitpicdepth}%
    \ifnum\value{thesisfitpicdepth}=1
      \ifthesisfitpics \let\thesisfitnext\thesisfitopen \else \let\thesisfitnext\relax \fi
    \else
      \let\thesisfitnext\relax
    \fi
    \thesisfitnext}
  \AfterEndEnvironment{tikzpicture}{%
    \ifnum\value{thesisfitpicdepth}=1
      \ifthesisfitpics \let\thesisfitnext\thesisfitclose \else \let\thesisfitnext\relax \fi
    \else
      \let\thesisfitnext\relax
    \fi
    \thesisfitnext
    \addtocounter{thesisfitpicdepth}{-1}}
\fi

% When a leanlisting is used without an optional [..] argument, LaTeX's optional
% argument lookahead (\@ifnextchar[) skips spaces across the line break and eats
% the leading indentation of the first content line. Replace the lookahead with a
% \futurelet check that does not skip spaces, so the first line keeps its indent.
\expandafter\let\expandafter\leanlisting@worker\csname\string\leanlisting\endcsname
\def\leanlisting{\futurelet\@let@token\leanlisting@optcheck}
\def\leanlisting@optcheck{%
  \ifx\@let@token[%
    \let\leanlisting@next\leanlisting@worker
  \else
    \def\leanlisting@next{\leanlisting@worker[]}%
  \fi
  \leanlisting@next}

% quantities are absolute: the two display font sizes, the lift that pulls the

% A4, so scale them by that factor to keep the head in proportion.

  \newcommand{\svennHeadDrop}{10mm}
\fi

% Venn geometry lengths — computed once \textwidth is known
% Left circle bleeds ~9% of textwidth into the inner (left) margin.
\newlength{\vennR}       % circle radius
\newlength{\vennLx}      % x of left circle centre
\newlength{\vennRx}      % x of right circle centre
\newlength{\vennMidX}    % x where "CHAPTER" is centred (left boundary of right circle)
\newlength{\vennCY}      % y of circle centres (negative = below TikZ origin)
\newlength{\vennNumY}    % y baseline of the chapter number

% Small Venn geometry for starred chapters (title overlaid on small circles)
\newlength{\svennR}      % small circle radius
\newlength{\svennLx}     % x of left (dark) circle centre
\newlength{\svennRx}     % x of right (light) circle centre
\newlength{\svennCY}     % y of small circle centres (positive = above the title baseline)

\AtBeginDocument{%
  \setlength{\vennR}    {\dimexpr 25\textwidth/100\relax}%
  % Left centre: the circle bleeds across the whole inner margin and stops
  % 2.5mm short of the paper edge, whatever that margin is (chapters always
  % open on a recto, so \oddsidemargin is the relevant one).
  \setlength{\vennLx}   {\dimexpr\vennR - \oddsidemargin - 1in + 2.5mm\relax}%
  \setlength{\vennRx}   {\dimexpr\vennLx + \textwidth/100*37\relax}%
  % Colour-split point = left boundary of right circle
  \setlength{\vennMidX} {\dimexpr\vennRx - \vennR\relax}%
  % Circles start just below the TikZ origin (top of bounding box)
  \setlength{\vennCY}   {\dimexpr -\vennR - 4mm\relax}%
  % Number baseline: ~0.6R above circle centre → sits in upper quarter of right circle
  \setlength{\vennNumY} {\dimexpr\vennCY + 6\vennR/10\relax}%
  % --- Small Venn (starred chapters), margin-aligned: no left bleed ---
  % Circles sit just ABOVE the title (which stays fully readable on the cream page).
  \setlength{\svennR}   {\ifThesisAfive 3mm\else 4mm\fi}%  % small radius
  \setlength{\svennLx}  {-.25\svennR}%                          % dark circle left edge at the text margin (x=0)
  \setlength{\svennRx}  {\dimexpr\svennLx + 14\svennR/10\relax}% % centres 1.4R apart (Venn overlap)
  \setlength{\svennCY}  {\ifThesisAfive 12mm\else 16mm\fi}%% centres above the title baseline (clear of the cap line)
}

\renewcommand{\@makechapterhead}[1]{%
  \vspace*{-\vennHeadLift}%
  \noindent
  \begingroup\thesisfitpicsfalse
  \tikzexternalexportnextfalse
  \begin{tikzpicture}%
    % Bounding box: the left circle bleeds past x=0 into the inner margin (intentional).
    % Top set to 30mm to accommodate the large number overflowing above the circles.
    \useasboundingbox
      (0pt, \dimexpr\vennCY - \vennR - 5mm\relax)
      rectangle (\textwidth, 20mm);
    % --- Circles ---
    \fill[venn-dark]  (\vennLx, \vennCY) circle (\vennR);
    \fill[venn-light, fill opacity=.95] (\vennRx, \vennCY) circle (\vennR);
    % --- "CHAPTER" in light colour, clipped to left circle ---
    \begin{scope}
      \clip (\vennLx, \vennCY) circle (\vennR);
      \node[anchor=center] at (\vennMidX, \vennCY)
        {{\playfairvennword\bfseries\fontsize{\vennWordSize}{\vennWordLead}\selectfont
          \textcolor{venn-textlight}{CHAPTER}}};
    \end{scope}
    % --- "CHAPTER" in dark colour, clipped to right circle ---
    \begin{scope}
      \clip (\vennRx, \vennCY) circle (\vennR);
      \node[anchor=center] at (\vennMidX, \vennCY)
        {{\playfairvennword\bfseries\fontsize{\vennWordSize}{\vennWordLead}\selectfont
          \textcolor{venn-textdark}{CHAPTER}}};
    \end{scope}
    % --- Chapter number: top-right of lighter circle, half-overflows into outer margin ---
    \node[anchor=base] at (\dimexpr\vennRx + 7\vennR/10\relax, \vennNumY)
      {{\playfaircoverfamily\itshape\bfseries\fontsize{\vennNumSize}{\vennNumLead}\selectfont
        \textcolor{venn-dark}{\thechapter}}};
  \end{tikzpicture}%
  \endgroup
  \par\nobreak\vskip 8\p@
  {\playfaircoverfamily\bfseries\Huge\raggedright\textcolor{venn-dark}{#1}\par}%
  \nobreak\vskip 10\p@
}

% Starred chapters (\chapter*{...}): title overlaid on two small Venn circles
% (dark left, light right), in the same two-tone style as numbered chapters.
% The circles are small and the title sits on top of them, overflowing to the right.
\renewcommand{\@makeschapterhead}[1]{%
  \vspace*{\svennHeadDrop}%
  \noindent
  \begingroup\thesisfitpicsfalse
  \tikzexternalexportnextfalse
  \begin{tikzpicture}%
    % Title baseline sits at the TikZ origin (y=0); the circles float above it.
    \useasboundingbox
      (0pt, -8mm)
      rectangle (\textwidth, \dimexpr\svennCY + \svennR + 2mm\relax);
    % --- Small circles, above the title ---
    \fill[venn-dark]                    (\svennLx, \svennCY) circle (\svennR);
    \fill[venn-light, fill opacity=.9] (\svennRx, \svennCY) circle (\svennR);
    % --- Title on the baseline, fully readable on the cream page ---
    \node[anchor=base west, inner sep=0pt] at (0pt, 0pt)
      {{\playfaircoverfamily\bfseries\Huge\textcolor{venn-dark}{#1}}};
    % --- Two-tone accent where ascenders kiss the circles ---
    \begin{scope}
      \clip (\svennLx, \svennCY) circle (\svennR);
      \node[anchor=base west, inner sep=0pt] at (0pt, 0pt)
        {{\playfaircoverfamily\bfseries\Huge\textcolor{venn-textlight}{#1}}};
    \end{scope}
    \begin{scope}
      \clip (\svennRx, \svennCY) circle (\svennR);
      \node[anchor=base west, inner sep=0pt] at (0pt, 0pt)
        {{\playfaircoverfamily\bfseries\Huge\textcolor{venn-textdark}{#1}}};
    \end{scope}
  \end{tikzpicture}%
  \endgroup
  \par\nobreak\vskip 24\p@
}

% --- Page number tabs bleeding into the outer margin ---
% Outer margin width: paperwidth minus the text block and its left offset.
\newlength{\outermarginwidth}
\newlength{\halfoutermarginwidth}
\newlength{\pagenumboxwidth}
\newlength{\pagenumboxsep}
% The tab must fit three digits plus its own padding inside \pagenumboxwidth.
% A5's 14mm outer margin cannot take the A4 tab, so shrink both there.
\ifThesisAfive
  \setlength{\pagenumboxsep}{5pt}
  \newcommand{\pagenumboxfont}{\fontsize{10}{12}}
\else
  \setlength{\pagenumboxsep}{10pt}
  \newcommand{\pagenumboxfont}{\fontsize{14}{16}}
\fi
\AtBeginDocument{%
  \setlength{\outermarginwidth}{\dimexpr\paperwidth - \textwidth - \oddsidemargin - 1in\relax}%
  \setlength{\halfoutermarginwidth}{0.5\outermarginwidth}%
  \setlength{\pagenumboxwidth}{0.85\outermarginwidth}%
}

% Odd pages (right-hand): used in [LO] slot.
% \rlap starts at the LEFT text margin; \hspace* skips ahead so the colorbox
% (width pagenumboxwidth, a fraction of the outer margin) ends flush at the
% right page edge.
\newcommand{\pagenumboxR}[1]{%
  \rlap{%
    \hspace*{\dimexpr\textwidth + \outermarginwidth - \pagenumboxwidth\relax}%
    {%
      \setlength{\fboxsep}{\pagenumboxsep}%
      \colorbox{gray!10}{%
        \makebox[\dimexpr\pagenumboxwidth - 2\pagenumboxsep\relax][c]{%
          \pagenumboxfont\sffamily\bfseries\itshape\selectfont
          \color{gray}#1%
        }%
      }%
    }%
  }%
}

% Even pages (left-hand): used in [LE] slot.
% \llap right-anchors at the LEFT text margin (= outermarginwidth from page edge).
% Content = colorbox(pagenumboxwidth) + hspace*(outermarginwidth - pagenumboxwidth)
% = outermarginwidth total, so the colorbox's left edge lands at the page edge.
\newcommand{\pagenumboxL}[1]{%
  \llap{%
    {%
        \setlength{\fboxsep}{\pagenumboxsep}%
        \colorbox{gray!10}{%
          \makebox[\dimexpr\pagenumboxwidth - 2\pagenumboxsep\relax][c]{%
            \pagenumboxfont\sffamily\bfseries\itshape\selectfont
            \color{gray}#1%
          }%
        }%
      }%
    \hspace*{\dimexpr\outermarginwidth - \pagenumboxwidth\relax}%
  }%
}

% --- Footer: page-number tab on the outer edge only, no rule ---
\pagestyle{fancy}
\fancyhf{}
\fancyfoot[LO]{\pagenumboxR{\thepage}}
\fancyfoot[LE]{\pagenumboxL{\thepage}}
\renewcommand{\footrulewidth}{0pt}
\renewcommand{\headrulewidth}{0pt}
\fancypagestyle{plain}{%
  \fancyhf{}
  \fancyfoot[LO]{\pagenumboxR{\thepage}}
  \fancyfoot[LE]{\pagenumboxL{\thepage}}
  \renewcommand{\footrulewidth}{0pt}
  \renewcommand{\headrulewidth}{0pt}
}
