% Phantom counters: each env type gets its own counter whose \the* macro

% Helper: top-left accent bar drawn inside an overlay (TikZ context only).
\newcommand{\tcbenvtopbar}[1]{%
  \draw[line width=2pt, color=#1, line cap=butt]
    (frame.north west) -- ([xshift=6mm]frame.north west);
}

\newcommand{\tcbenvleanptr}[2]{%
  \if\relax\detokenize{#1}\relax\else
    \node[anchor=south east, inner sep=3pt, yshift=-4pt]
      at (frame.north east)
      {%
        \scriptsize
        \if\relax\detokenize{#2}\relax
          \ttfamily\color{beige!25!cover-cream}#1%
        \else
          \begingroup
          \hypersetup{urlcolor=beige!25!cover-cream}%
          \leancodeptr{#1}{#2}%
          \endgroup
        \fi
      };%
  \fi
}

% Render lean pointer pairs stacked in top-right corner.
% #1 = comma-separated lean names, #2 = comma-separated URLs (same count, or empty).
% Empty #1 = nothing rendered. Single entry = backward compatible with \tcbenvleanptr.
\ExplSyntaxOn
\seq_new:N \l__leanptr_names_seq
\seq_new:N \l__leanptr_urls_seq
\tl_new:N  \l__leanptr_name_tl
\tl_new:N  \l__leanptr_url_tl
\int_new:N \l__leanptr_n_int
\int_new:N \l__leanptr_i_int

\cs_new_protected:Nn \__leanptr_render_pair:nn {
  \tl_if_blank:nTF { #2 } {
    \ttfamily\color{beige!25!cover-cream}#1%
  }{
    \begingroup\hypersetup{urlcolor=beige!25!cover-cream}%
    \leancodeptr{#1}{#2}%
    \endgroup
  }
}
\cs_generate_variant:Nn \__leanptr_render_pair:nn { VV }
\cs_generate_variant:Nn \seq_set_split:Nnn { Nne }

\NewDocumentCommand \tcbenvleanptrlist { m m } {
  \tl_if_blank:nF { #1 } {
    \seq_set_split:Nnn \l__leanptr_names_seq { , } { #1 }
    \seq_set_split:Nne \l__leanptr_urls_seq  { , } { #2 }
    \int_set:Nn \l__leanptr_n_int { \seq_count:N \l__leanptr_names_seq }
    \node[anchor=south~east, inner~sep=3pt, yshift=-3pt] at (frame.north~east) {%
      \scriptsize
      \int_step_inline:nn { \l__leanptr_n_int } {
        \int_compare:nNnT { ##1 } > { 1 } { \textcolor{beige!25!cover-cream}{,}~ }
        \tl_set:Nx \l__leanptr_name_tl { \seq_item:Nn \l__leanptr_names_seq { ##1 } }
        \tl_set:Nx \l__leanptr_url_tl  { \seq_item:Nn \l__leanptr_urls_seq  { ##1 } }
        \tl_trim_spaces:N \l__leanptr_name_tl
        \tl_trim_spaces:N \l__leanptr_url_tl
        \__leanptr_render_pair:VV \l__leanptr_name_tl \l__leanptr_url_tl
      }%
    };%
  }
}
\ExplSyntaxOff

    \end{tcolorbox}%
  }
