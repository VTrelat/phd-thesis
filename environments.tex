% ---------------------------------------------------------------------------
% Shared display counter, reset by section.
% ---------------------------------------------------------------------------
\newcounter{thmcounter}[section]
\renewcommand{\thethmcounter}{\thesection.\arabic{thmcounter}}

% Phantom counters: each env type gets its own counter whose \the* macro
% delegates to \thethmcounter. \refstepcounter on the phantom sets
% \@currentlabel so that \label inside the env body resolves to the right
% type name in cleveref, without affecting the displayed number.
\newcounter{thmtype}[section]  \renewcommand{\thethmtype}{\thethmcounter}
\newcounter{lemtype}[section]  \renewcommand{\thelemtype}{\thethmcounter}
\newcounter{cortype}[section]  \renewcommand{\thecortype}{\thethmcounter}
\newcounter{deftype}[section]  \renewcommand{\thedeftype}{\thethmcounter}
\newcounter{extype}[section]   \renewcommand{\theextype}{\thethmcounter}
\newcounter{remtype}[section]  \renewcommand{\theremtype}{\thethmcounter}

% Helper: top-left accent bar drawn inside an overlay (TikZ context only).
\newcommand{\tcbenvtopbar}[1]{%
  \draw[line width=2pt, color=#1, line cap=butt]
    (frame.north west) -- ([xshift=6mm]frame.north west);
}

\newcommand{\tcbenvleanptr}[2]{%
  \if\relax\detokenize{#1}\relax\else
    \node[anchor=south east, inner sep=3pt]
      at (frame.north east)
      {%
        \footnotesize
        \if\relax\detokenize{#2}\relax
          \ttfamily\color{beige!25!cover-cream}#1%
        \else
          \begingroup
          \hypersetup{urlcolor=beige!25!cover-cream}%
          \leancodeptr{#1}{#2}%
          \endgroup
        \fi
      };%
  \fi
}

% ---------------------------------------------------------------------------
% Base style: beige background, no frame, matches leanlisting aesthetic.
% ---------------------------------------------------------------------------
\tcbset{
  envbase/.style={
    enhanced,
    breakable,
    arc=5pt,
    sharp corners=west,
    boxrule=0pt,
    frame hidden,
    colback=beige!30,
    left=8pt, right=8pt, top=4pt, bottom=4pt,
    before upper={\setlength{\parskip}{4pt}},
  },
  % envaccent={color}: tinted bg, left bar, floating title tab (solid color, white text).
  % Top bar and Lean name are handled per-env in overlay unbroken and first.
  envaccent/.style n args={1}{
    colback=#1!5,
    top=10pt,
    borderline west={2pt}{0pt}{#1},
    fonttitle=\small\sffamily\bfseries\color{white},
    attach boxed title to top left={yshift={-\dimexpr\tcboxedtitleheight/2\relax}, xshift=6mm},
    boxed title style={
      enhanced,
      colback=#1,
      colframe=#1,
      arc=2pt,
      boxrule=0pt,
      left=4pt, right=4pt, top=1pt, bottom=1pt,
    },
  },
}

% THEOREM  (pink)
% ---------------------------------------------------------------------------
\NewTColorBox{theorembox}{ m m m }{%
  envbase, envaccent={pink},
  before upper={\refstepcounter{thmtype}\setlength{\parskip}{4pt}},
  title={%
    \stepcounter{thmcounter}%
    Theorem~\thethmcounter%
    \if\relax\detokenize{#1}\relax\else~\textup{(#1)}\fi
  },
  overlay unbroken and first={%
    \tcbenvtopbar{pink}%
    \tcbenvleanptr{#2}{#3}%
  },
}

  \edef\thesisleanurl{\detokenize{#3}}%

}{\endtheorembox}
